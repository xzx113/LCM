#Regular expression matching code

import pandas as pd
import re

# ===================== 1. 催收力度（六类行为标志） =====================
L1_patterns = [
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L2_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L3_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L4_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L5_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD$)', re.I),
]
L6_patterns = [
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]

def get_collection_flags(text):
    """返回六类催收行为的0/1标志"""
    if not isinstance(text, str):
        text = str(text)
    return {
        'L1_attitude': int(any(p.search(text) for p in L1_patterns)),
        'L2_friend': int(any(p.search(text) for p in L2_patterns)),
        'L3_family': int(any(p.search(text) for p in L3_patterns)),
        'L4_work': int(any(p.search(text) for p in L4_patterns)),
        'L5_visit': int(any(p.search(text) for p in L5_patterns)),
        'L6_lawsuit': int(any(p.search(text) for p in L6_patterns)),
    }

# ===================== 2. 催收频次等级（优化版） =====================
def get_frequency_level(text):
    if not isinstance(text, str):
        text = str(text)
    # 数字匹配（限定上下文）
    digit_patterns = [
        re.compile(r'(?:SENSITIVE_WORD)?\s*(?:SENSITIVE_WORD)\s*(\d+)\s*(?:SENSITIVE_WORD)?\s*SENSITIVE_WORD', re.I),
        re.compile(r'(?:SENSITIVE_WORD)?\s*(\d+)\s*(?:SENSITIVE_WORD)?\s*SENSITIVE_WORD', re.I),
        re.compile(r'SENSITIVE_WORD?(\d+)\s*SENSITIVE_WORD', re.I),
        re.compile(r'SENSITIVE_WORD(\d+)\s*SENSITIVE_WORD', re.I),
        re.compile(r'(\d+)\s*SENSITIVE_WORD', re.I),
    ]
    max_num = 0
    for pat in digit_patterns:
        m = pat.search(text)
        if m:
            # 排除自我否定语境（简略版）
            start, end = m.span()
            preceding = text[max(0, start-15):start]
            if re.search(r'(SENSITIVE_WORD)', preceding, re.I):
                continue
            num = int(m.group(1))
            if num > max_num:
                max_num = num
    if max_num > 0:
        if max_num <= 5:
            return 1
        elif max_num <= 15:
            return 2
        else:
            return 3
    # 模糊关键词
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 3
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 2
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 1
    # 兜底：“SENSITIVE_WORD”无数字 → 2；“SENSITIVE_WORD”无数字 → 1
    if re.search(r'(SENSITIVE_WORD).*打', text, re.I) or re.search(r'(SENSITIVE_WORD).*SENSITIVE_WORD', text, re.I):
        return 2
    if re.search(r'SENSITIVE_WORD.*SENSITIVE_WORD', text, re.I) or re.search(r'SENSITIVE_WORD.*SENSITIVE_WORD', text, re.I):
        return 1
    return 0

# ===================== 3. 敏感词（B1A，） =====================
core_sensitive_words = [
    SENSITIVE_WORD
]
escaped_words = [re.escape(w) for w in core_sensitive_words]
sensitive_pattern = re.compile('|'.join(escaped_words), re.I)

def get_sensitive_density(text):
    """返回敏感词密度（次数/总字符数 * 100）"""
    if not isinstance(text, str):
        text = str(text)
    if len(text) == 0:
        return 0.0
    count = len(sensitive_pattern.findall(text))
    return (count / len(text)) * 100

# ===================== 主处理 =====================
input_path = r"基础数据.xlsx"
df = pd.read_excel(input_path)

# 确保有Text列
if 'Text' not in df.columns:
    raise ValueError("Excel文件中缺少'Text'列，请检查")

# 提取特征
flags_list = []
freq_list = []
sens_list = []
for text in df['Text']:
    flags = get_collection_flags(text)
    flags_list.append(flags)
    freq_list.append(get_frequency_level(text))
    sens_list.append(get_sensitive_density(text))

# 将flags字典拆分为多个列
flags_df = pd.DataFrame(flags_list)
result_df = pd.concat([df, flags_df, 
                       pd.DataFrame({'催收频次等级': freq_list}),
                       pd.DataFrame({'敏感词密度': sens_list})], axis=1)

# 将中文列名改为英文
result_df = result_df.rename(columns={
    '催收频次等级': 'collection_freq_level',
    '敏感词密度': 'sensitive_word_density'
})

# 保存
output_path = r"基础数据_特征标记.xlsx"
result_df.to_excel(output_path, index=False)
print(f"处理完成！共处理 {len(df)} 条记录。")
print("新增列：L1_attitude, L2_friend, L3_family, L4_work, L5_visit, L6_lawsuit, collection_freq_level, sensitive_word_density")

client = OpenAI(api_key=API_KEY, base_url=BASE_URL)

# ===================== Prompt ======================
LCM_PROMPT = """
你是一个文本分析工具。请严格按照下面的要求分析用户输入的文本，**只输出一行 JSON**，不要输出任何其他解释或标记。

### 任务
统计文本中以下两类语言特征的数量：

1. **负向形容词（ADJ）**：描述催收方/平台的负面**稳定特质**，例如：无耻、恶心、暴力、恶劣、垃圾（作形容词时）、可恶。
   - 需要判断指向对象是否为「催收方、平台、公司、你们、美团」等（计入 ADJ_to_creditor）。
   - **注意**：描述“情绪状态”的词（如崩溃、愤怒、绝望、烦躁）无论是否被副词修饰，一律归入 SV（见下方），不归入 ADJ。

2. **负向状态动词（SV）**：表达因对方行为或处境而引发的**负面情绪状态**。例如：恨、讨厌、反感、愤怒、崩溃、绝望、受不了、烦、生气、痛苦。
   - **不需要判断指向对象**，只要词语本身表达了负面情绪状态，即计入 SV_negative_total。
   - 包含“让/使/令…崩溃/愤怒”这类短语中的核心词。
   - **归属优先级**：当一个词既可作形容词又可作情绪动词时（如“崩溃”“愤怒”），一律归入 SV，不归入 ADJ。

### 不需要统计的类别
- 描述性行动动词（如“打电话”“还钱”）
- 解释性行动动词（如“骚扰”“威胁”“欺骗”）
- 其他中性或正向词汇

### 输出格式（严格 JSON）
{{
  "ADJ_total": 数字,
  "ADJ_to_creditor": 数字,
  "SV_negative_total": 数字
}}

### Few-shot 示例

示例1：
文本：你们平台真是太无耻了，我恨你们。
输出：
{{
  "ADJ_total": 1,
  "ADJ_to_creditor": 1,

  "SV_negative_total": 1
}}

示例3：
文本：太可恶了！真受不了你们。
输出：
{{
  "ADJ_total": 1,

示例4：
文本：你们这样让我很愤怒，非常讨厌。
输出：

示例5：
文本：你们太让人崩溃了，我彻底愤怒了。

### 待分析的文本
{text}
"""


def extract_json_from_response(response_text: str) -> dict:
    """从LLM返回的文本中提取JSON对象"""
    try:
        return json.loads(response_text.strip())
    except:
        pass
    match = re.search(r'\{[^{}]*\}', response_text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group())
        except:
            pass
    # 默认返回移除了 ADJ_intense_to_creditor
    return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}


def get_lcm_counts(text: str) -> dict:
    """调用LLM获取LCM统计计数"""
    if not text or len(text.strip()) < 10:
        return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}
    if len(text) > MAX_TEXT_LENGTH:
        text = text[:MAX_TEXT_LENGTH]

    prompt = LCM_PROMPT.format(text=text)

    for attempt in range(RETRY_TIMES):
        try:
            response = client.chat.completions.create(
                model=MODEL_NAME,
                messages=[{"role": "user", "content": prompt}],
                temperature=0,
                max_tokens=300,
            )
            content = response.choices[0].message.content
            counts = extract_json_from_response(content)
          
            for key in ["ADJ_total", "ADJ_to_creditor", "SV_negative_total"]:
                if key not in counts:
                    counts[key] = 0
            return counts
        except Exception as e:
            print(f"  请求失败 (尝试 {attempt + 1}/{RETRY_TIMES}): {str(e)[:50]}")
            time.sleep(2)
    return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}


def compute_b1b(counts: dict, text: str) -> float:
    """计算 B1B 分数，"""
    total_chars = len(text)
    if total_chars == 0:
        return 0.0
    # 只保留 ADJ_to_creditor * 4 和 SV * 3
    weighted_sum = counts["ADJ_to_creditor"] * 4 + counts["SV_negative_total"] * 3
    return (weighted_sum / total_chars) * 100


def main():
    # 请根据实际数据位置修改
    input_file = "./data/打分数据集_cleaned.xlsx"  
    df = pd.read_excel(input_file)

        print(f"错误：Excel文件中未找到列 '{text_column}'。可用的列有：{list(df.columns)}")

    results = []
    for idx, row in df.iterrows():
        text = str(row[text_column]) if pd.notna(row[text_column]) else ""
        if len(text.strip()) < 10:
            print(f"跳过第{idx + 1}行：文本过短")
            results.append({
                "客户ID": row.get("客户ID", idx + 1),
                "ADJ_total": 0,
                "ADJ_to_creditor": 0,
                "SV_negative_total": 0,
                "B1B_score": 0,
                "text_length": len(text)
            })
            continue

        print(f"处理第{idx + 1}行，文本长度: {len(text)}")
        counts = get_lcm_counts(text)
        b1b = compute_b1b(counts, text)
        results.append({
            "客户ID": row.get("客户ID", idx + 1),
            "ADJ_total": counts["ADJ_total"],
            "ADJ_to_creditor": counts["ADJ_to_creditor"],
            "SV_negative_total": counts["SV_negative_total"],
            "B1B_score": b1b,
            "text_length": len(text)
        })
        time.sleep(SLEEP_TIME)

  
    output_file = "./output/LCM统计结果.xlsx"  
    pd.DataFrame(results).to_excel(output_file, index=False)
    print(f"\n✅ 处理完成！共处理 {len(results)} 条文本，结果保存至：{output_file}")


if __name__ == "__main__":
    main()



#Code for the dose-response analysis in RQ A (three-group gradient comparison)

import pandas as pd
import numpy as np

import statsmodels.api as sm
import math
import warnings

# ==================== 1. 读取数据 ====================
file_path = r"XX"
df = pd.read_excel(file_path)

print("=" * 70)

# ==================== 2. 构造三组 ====================
# 完整信号：三指标均>0
# 残缺信号：至少一个>0，但不是全部>0
# 无信号：全为0

df['has_abstract'] = ((df['ADJ_Ratio'] > 0) | (df['SV_Ratio'] > 0)).astype(int)
df['has_channel'] = (df['sensitive_word_density'] > 0).astype(int)

df['signal_tier'] = '无信号'
df.loc[(df['has_abstract'] == 1) & (df['has_channel'] == 0), 'signal_tier'] = '残缺信号(IF only)'
df.loc[(df['has_abstract'] == 0) & (df['has_channel'] == 1), 'signal_tier'] = '残缺信号(THEN only)'
df.loc[(df['has_abstract'] == 1) & (df['has_channel'] == 1), 'signal_tier'] = '完整信号'

# 合并残缺信号为一个组（可选）
df['signal_group'] = df['signal_tier'].replace({
    '残缺信号(IF only)': '残缺信号',
    '残缺信号(THEN only)': '残缺信号'
})


# ==================== 3. 辅助函数 ====================
def cohens_h(p1, p2):
    if p1 == p2:
        return 0.0
    h1 = 2 * math.asin(math.sqrt(p1))
    h2 = 2 * math.asin(math.sqrt(p2))
    return h1 - h2


def risk_ratio(p1, p2):
    if p2 == 0:
        return np.nan
    return p1 / p2


def risk_difference_ci(p1, n1, p2, n2, alpha=0.05):
    ci1_low, ci1_high = proportion_confint(int(n1 * p1), n1, alpha=alpha, method='wilson')
    ci2_low, ci2_high = proportion_confint(int(n2 * p2), n2, alpha=alpha, method='wilson')
    se = np.sqrt((ci1_high - ci1_low) ** 2 / 4 + (ci2_high - ci2_low) ** 2 / 4)
    diff = p1 - p2
    z = 1.96
    return diff, diff - z * se, diff + z * se


# ==================== 4. 分组统计 ====================
tier_order = ['无信号', '残缺信号', '完整信号']
summary = []
for tier in tier_order:
    group = df[df['signal_group'] == tier]
    n = len(group)
    complaints = group['Spillover_flag'].sum()
    p = group['Spillover_flag'].mean()
    ci_low, ci_high = proportion_confint(complaints, n, alpha=0.05, method='wilson')
    summary.append({
        'tier': tier,
        'n': n,
        'complaints': complaints,
        'p': p,
        'ci_low': ci_low,
        'ci_high': ci_high
    })

print("\n【三组原始统计】")
print("-" * 70)

print("-" * 70)
for s in summary:
    ci_str = f"({s['ci_low']:.3f}, {s['ci_high']:.3f})"
    print(f"{s['tier']:<12} {s['n']:>10} {s['complaints']:>10} {s['p']:>12.3f} {ci_str:>20}")

# ==================== 5. 两两对比 ====================
print("\n" + "=" * 70)
print("【两两对比】")

    p1, p2 = g1['Spillover_flag'].mean(), g2['Spillover_flag'].mean()

    rr_high_vs_low = risk_ratio(p2, p1) if p1 > 0 else np.nan
    diff, diff_low, diff_high = risk_difference_ci(p1, n1, p2, n2)
    h = cohens_h(p1, p2)

    results.append({
        '对比': f"{tier1} → {tier2}",
        'p1': p1, 'n1': n1,
        'p2': p2, 'n2': n2,
        '风险差': diff,
        '风险差CI_low': diff_low,
        '风险差CI_high': diff_high,
        '风险比_RR': rr_high_vs_low,
        'cohens_h': h
    })

print(
    f"\n{'对比':<20} {'投诉率1':>12} {'投诉率2':>12} {'风险差':>12} {'95% CI':>20} {'风险比(RR)':>12} {'Cohen_h':>12}")
print("-" * 105)

# ==================== 6. 卡方趋势检验 ====================
print("\n" + "=" * 70)
print("【剂量-反应趋势检验】")

# ==================== 7. 汇总（论文可直接引用） ====================
print("\n" + "=" * 70)
print("【论文可直接引用的结果】")
print("=" * 70)

no_signal = df[df['signal_group'] == '无信号']

# 风险比及CI（完整 vs 无信号）
rr = p_full / p_none if p_none > 0 else np.nan
if not np.isnan(rr) and rr > 0:
    log_rr = math.log(rr)

def compute_smd(df, treatment_col, cols):
    treated = df[df[treatment_col] == 1]
    control = df[df[treatment_col] == 0]
    smd_dict = {}
    for col in cols:
        if col in df.columns:
            m1, m2 = treated[col].mean(), control[col].mean()
            v1, v2 = treated[col].var(), control[col].var()
            pooled = np.sqrt((v1 + v2) / 2)
            smd_dict[col] = abs((m1 - m2) / pooled) if pooled > 0 else 0
    return smd_dict

def bootstrap_att(treatment_out, control_out, n_bootstrap=1000, seed=42):
    np.random.seed(seed)
    n_pairs = len(treatment_out)
    atts = []
    for _ in range(n_bootstrap):
        idx = np.random.choice(n_pairs, size=n_pairs, replace=True)
        t_sample = treatment_out[idx]
        c_sample = control_out[idx]
        atts.append(t_sample.mean() - c_sample.mean())
    lower = np.percentile(atts, 2.5)
    upper = np.percentile(atts, 97.5)
    return lower, upper, atts

# 10. Rosenbaum敏感性分析（表D）
# ============================================================
gamma_values = [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]


def rosenbaum_sensitivity(t_out, c_out, gamma_values):
    diffs = t_out - c_out
    n = len(diffs)
    abs_diffs = np.abs(diffs)
    ranks = np.argsort(abs_diffs) + 1
    signs = np.sign(diffs)
    results = []
    for gamma in gamma_values:
        w_plus = np.sum(ranks * (signs == 1))
        w_minus = np.sum(ranks * (signs == -1))
        w_stat = min(w_plus, w_minus)
        if n > 0 and np.var(diffs) > 0:
            try:
                _, p_val = wilcoxon(diffs, correction=True)
            except:
                p_val = 1.0
        else:
            p_val = 1.0
        p_upper = min(1.0, p_val * gamma)
        results.append({'gamma': gamma, 'p_upper': p_upper})
    return results

# ===================== 核心工具函数 =====================
def split_overlong_text(text: str, max_len: int = MAX_TEXT_LENGTH) -> str:
    """
    自动切分超长文本 
    """
    text = str(text).strip()
    if len(text) <= max_len:
        return text
    # 超长则截断，保留前max_len字符
    return text[:max_len]

def get_weights_by_length(text: str) -> dict:
    word_count = len(text)
    return weights_300_1000 if word_count <= 1000 else weights_full

def format_weights_text(weights: dict) -> str:
    sorted_items = sorted(weights.items(), key=lambda x: x[1], reverse=True)
    return "\n".join([f"- {word}（权重 {round(weight, 1)}）" for word, weight in sorted_items])

def clean_response(res: str) -> str:
    """清洗模型响应，去除标点、空格、特殊字符"""
    return re.sub(r'[^\u4e00-\u9fa5]', '', res.strip())

def map_score(response: str) -> int:
    """将文本响应映射为分数"""
    option_map = {
        "非常不同意": 1, "不同意": 2, "中立": 3, "同意": 4, "非常同意": 5
    }
    return option_map.get(response, 3)  # 默认中立

# ===================== 评分函数 =====================
def llm_score(text: str, item: str, is_neuro: bool) -> int:
    """统一评分函数，神经质/尽责性二合一"""
    text = split_overlong_text(text)  # 🔥 关键：自动处理超长文本
    weights = get_weights_by_length(text)
    weights_text = format_weights_text(weights)

    # 神经质 Prompt
    if is_neuro:
        prompt = f"""你的任务是基于下方提供的客户对话文本，回答人格量表题目。请扮演客户，反映其人格特质。
【贷后场景重点词汇权重指引】
{weights_text}
高权重词汇=更高神经质倾向。

【评分规则】
非常不同意/不同意/中立/同意/非常同意

题目：{item}
文本：{text}

【严格指令】只输出5个选项之一，无其他内容！"""
    # 尽责性 Prompt
    else:
        prompt = f"""你的任务是基于下方提供的客户对话文本，回答人格量表题目。请扮演客户，反映其人格特质。
【贷后尽责性评估指引】
1. 明确近期还款承诺（本月/具体日期）
2. 主动咨询分期
符合则高分，推诿则低分。

【评分规则】
非常不同意/不同意/中立/同意/非常同意

题目：{item}
文本：{text}

【严格指令】只输出5个选项之一，无其他内容！"""

    # 循环重试（替代递归，杜绝卡死）
    for _ in range(RETRY_TIMES):
        try:
            response = client.chat.completions.create(
                model=MODEL_NAME,
                messages=[{"role": "user", "content": prompt}],
                temperature=0,
                max_tokens=10,
            )
            res = clean_response(response.choices[0].message.content)
            return map_score(res)
        except Exception as e:
            print(f"评分失败，重试中... 错误：{str(e)[:50]}")
            time.sleep(2)
    return 3  # 最终失败返回中立分

# ===================== 分数计算 =====================
def calc_dim_score(scores):
    avg = sum(scores) / len(scores)
    return round(avg * 25 - 25, 2)

# ===================== 主程序 =====================
def run_dual_dimension():
    input_file = r" /data/input.xlsx"
    df = pd.read_excel(input_file)

    if "客户ID" not in df.columns:
        df["客户ID"] = range(1, len(df) + 1)

    results = []
    for idx, row in df.iterrows():
        cid = row["客户ID"]
        text1 = str(row["text1"]) if pd.notna(row["text1"]) else ""
        text2 = str(row["text2"]) if pd.notna(row["text2"]) else ""

        if len(text1.strip()) < 10 or len(text2.strip()) < 10:
            print(f"客户{cid}文本过短，跳过")
            continue

        print(f"=== 处理客户 {cid} ===")

        # 第一通对话
        neuro_scores1 = [llm_score(text1, item, True) for item in neuro_items]
        cons_scores1 = [llm_score(text1, item, False) for item in cons_items]
        time.sleep(SLEEP_TIME)

        # 第二通对话
        neuro_scores2 = [llm_score(text2, item, True) for item in neuro_items]
        cons_scores2 = [llm_score(text2, item, False) for item in cons_items]

        # 计算最终分
        neuro1, neuro2 = calc_dim_score(neuro_scores1), calc_dim_score(neuro_scores2)
        cons1, cons2 = calc_dim_score(cons_scores1), calc_dim_score(cons_scores2)

        results.append({
            "客户ID": cid, "神经质_1": neuro1, "神经质_2": neuro2,
            "尽责性_1": cons1, "尽责性_2": cons2,
        })
        print(f"  神经质: {neuro1} / {neuro2} | 尽责性: {cons1} / {cons2}")

    # 保存结果
    output_file = r" /data/input.xlsx"
    pd.DataFrame(results).to_excel(output_file, index=False)
    print(f"\n✅ 全部完成！结果已保存：{output_file}")

if __name__ == "__main__":
run_dual_dimension()
