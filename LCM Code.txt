import pandas as pd

# ===================== 1. 催收力度（六类行为标志） =====================
L1_patterns = [
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L2_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L3_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L4_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L5_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD$)', re.I),
]
L6_patterns = [
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]

def get_collection_flags(text):
    """返回六类催收行为的0/1标志"""
    if not isinstance(text, str):
        text = str(text)
    return {
        'L1_attitude': int(any(p.search(text) for p in L1_patterns)),
        'L2_friend': int(any(p.search(text) for p in L2_patterns)),
        'L3_family': int(any(p.search(text) for p in L3_patterns)),
        'L4_work': int(any(p.search(text) for p in L4_patterns)),
        'L5_visit': int(any(p.search(text) for p in L5_patterns)),
        'L6_lawsuit': int(any(p.search(text) for p in L6_patterns)),
    }

# ===================== 2. 催收频次等级（优化版） =====================
def get_frequency_level(text):
    if not isinstance(text, str):
        text = str(text)
    # 数字匹配（限定上下文）
    digit_patterns = [
        re.compile(r'(?:SENSITIVE_WORD)?\s*(?:SENSITIVE_WORD)\s*(\d+)\s*(?:SENSITIVE_WORD)?\s*SENSITIVE_WORD', re.I),
        re.compile(r'(?:SENSITIVE_WORD)?\s*(\d+)\s*(?:SENSITIVE_WORD)?\s*SENSITIVE_WORD', re.I),
        re.compile(r'SENSITIVE_WORD?(\d+)\s*SENSITIVE_WORD', re.I),
        re.compile(r'SENSITIVE_WORD(\d+)\s*SENSITIVE_WORD', re.I),
        re.compile(r'(\d+)\s*SENSITIVE_WORD', re.I),
    ]
    max_num = 0
    for pat in digit_patterns:
        m = pat.search(text)
        if m:
            # 排除自我否定语境（简略版）
            start, end = m.span()
            preceding = text[max(0, start-15):start]
            if re.search(r'(SENSITIVE_WORD)', preceding, re.I):
                continue
            num = int(m.group(1))
            if num > max_num:
                max_num = num
    if max_num > 0:
        if max_num <= 5:
            return 1
        elif max_num <= 15:
            return 2
        else:
            return 3
    # 模糊关键词
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 3
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 2
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 1
    # 兜底：“SENSITIVE_WORD”无数字 → 2；“SENSITIVE_WORD”无数字 → 1
    if re.search(r'(SENSITIVE_WORD).*打', text, re.I) or re.search(r'(SENSITIVE_WORD).*SENSITIVE_WORD', text, re.I):
        return 2
    if re.search(r'SENSITIVE_WORD.*SENSITIVE_WORD', text, re.I) or re.search(r'SENSITIVE_WORD.*SENSITIVE_WORD', text, re.I):
        return 1
    return 0

# ===================== 3. 敏感词（B1A，） =====================
core_sensitive_words = [
    SENSITIVE_WORD
]
escaped_words = [re.escape(w) for w in core_sensitive_words]
sensitive_pattern = re.compile('|'.join(escaped_words), re.I)

def get_sensitive_density(text):
    """返回敏感词密度（次数/总字符数 * 100）"""
    if not isinstance(text, str):
        text = str(text)
    if len(text) == 0:
        return 0.0
    count = len(sensitive_pattern.findall(text))
    return (count / len(text)) * 100

input_path = r"基础数据.xlsx"

def extract_json_from_response(response_text: str) -> dict:
    """从LLM返回的文本中提取JSON对象"""
    try:
        return json.loads(response_text.strip())
    except:
        pass
    match = re.search(r'\{[^{}]*\}', response_text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group())
        except:
            pass
    # 默认返回移除了 ADJ_intense_to_creditor
    return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}


def get_lcm_counts(text: str) -> dict:
    """调用LLM获取LCM统计计数"""
    if not text or len(text.strip()) < 10:
        return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}
    if len(text) > MAX_TEXT_LENGTH:
        text = text[:MAX_TEXT_LENGTH]

    prompt = LCM_PROMPT.format(text=text)

    for attempt in range(RETRY_TIMES):
        try:
            response = client.chat.completions.create(
                model=MODEL_NAME,
                messages=[{"role": "user", "content": prompt}],
                temperature=0,
                max_tokens=300,
            )
            content = response.choices[0].message.content
            counts = extract_json_from_response(content)
          
            for key in ["ADJ_total", "ADJ_to_creditor", "SV_negative_total"]:
                if key not in counts:
                    counts[key] = 0
            return counts
        except Exception as e:
            print(f"  请求失败 (尝试 {attempt + 1}/{RETRY_TIMES}): {str(e)[:50]}")
            time.sleep(2)
    return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}


def compute_b1b(counts: dict, text: str) -> float:
    """计算 B1B 分数，"""
    total_chars = len(text)
    if total_chars == 0:
        return 0.0
    # 只保留 ADJ_to_creditor * 4 和 SV * 3
    weighted_sum = counts["ADJ_to_creditor"] * 4 + counts["SV_negative_total"] * 3
    return (weighted_sum / total_chars) * 100

# 构建KDTree，查询时一次返回前5个最近邻（可根据需要调整）

matched_pairs = []  # 存储 (处理组索引, 对照组索引) 元组
used_control_idx = set()  # 已使用的对照组在control_df中的位置

for i in range(len(treatment_df)):
    # 当前处理组的候选对照距离和索引
    dist_list = distances[i]  # 长度为 k_neighbors 的数组
    idx_list = indices[i]  # 长度为 k_neighbors 的数组

    # 按距离从小到大排序（已由KDTree返回，但保险起见再排一次）
    sorted_pairs = sorted(zip(dist_list, idx_list), key=lambda x: x[0])

    matched = False
    for dist, c_idx in sorted_pairs:
        if dist > caliper:
            break  # 距离超出卡尺，后面的更远，直接放弃
        if c_idx not in used_control_idx:
            # 找到可用对照
            matched_pairs.append((i, c_idx))
            used_control_idx.add(c_idx)
            matched = True
            break
    if not matched:
        # 该处理组未匹配到合适对照，丢弃
        print(f"处理组样本 {i} (ps={treatment_df.loc[i, 'ps']:.4f}) 未找到匹配，已丢弃")

# ==================== 6. 平衡性检验（SMD） ====================
def compute_smd(df, treatment_col, covariates):
    treated = df[df[treatment_col] == 1]
    control = df[df[treatment_col] == 0]
    smd_dict = {}
    for col in covariates:
        mean_t = treated[col].mean()
        mean_c = control[col].mean()
        var_t = treated[col].var()
        var_c = control[col].var()
        pooled_std = np.sqrt((var_t + var_c) / 2)
        smd = (mean_t - mean_c) / pooled_std if pooled_std > 0 else 0
        smd_dict[col] = abs(smd)
    return smd_dict

print(f"差异 (ATT): {diff:.3f}")

# 构建配对四格表的四个格子
a = ((matched_treatment['Spillover_flag'] == 0) & (matched_control['Spillover_flag'] == 0)).sum()
b = ((matched_treatment['Spillover_flag'] == 0) & (matched_control['Spillover_flag'] == 1)).sum()
c = ((matched_treatment['Spillover_flag'] == 1) & (matched_control['Spillover_flag'] == 0)).sum()
d = ((matched_treatment['Spillover_flag'] == 1) & (matched_control['Spillover_flag'] == 1)).sum()

print(f"\n配对四格表：")
print(f"         对照组未投诉  对照组投诉")
print(f"处理组未投诉    {a}          {b}")
print(f"处理组投诉      {c}          {d}")

# McNemar 检验只关注不一致的对子（b 和 c）
n = b + c  # 不一致对子的总数
if n == 0:
    print("警告：没有不一致的对子，无法计算 McNemar 检验 p 值")
else:
    # 精确双侧检验：在零假设下（b 和 c 等概率），P 值 = 2 * P(X <= min(b, c))
    # 其中 X ~ Binomial(n, 0.5)
    k = min(b, c)
    # 计算累积概率（双侧）
    p_value = 2 * binom.cdf(k, n, 0.5)
    # 如果 k 超过了 n/2，binom.cdf 会大于 0.5，导致 p_value > 1，需修正
    if p_value > 1:
        p_value = 2 * (1 - binom.cdf(k - 1, n, 0.5))

    print(f"\nMcNemar 精确检验 p 值: {p_value:.4f} {'（显著）' if p_value < 0.05 else '（不显著）'}")
# ==================== 8. 保存匹配结果 ====================
output_path = r"匹配结果_带卡尺3.xlsx"
matched_df.to_excel(output_path, index=False)
print(f"\n匹配结果已保存至 {output_path}")

import pandas as pd
import numpy as np
import lightgbm as lgb
from sklearn.model_selection import train_test_split
from sklearn.metrics import f1_score, precision_score, recall_score, roc_auc_score
import warnings
import os

warnings.filterwarnings('ignore')

OUTPUT_DIR = r"Your file path"
os.makedirs(OUTPUT_DIR, exist_ok=True)

# 读取数据
file_path = r"Your file path"
df = pd.read_excel(file_path)

print(f"全样本量: {len(df)}")
print(f"正样本（投诉）: {df['Spillover_flag'].sum()}，占比: {df['Spillover_flag'].mean():.2%}")

# 特征列
p_cols = ['Neuroticism', 'Conscientiousness', 'Agreeableness']
e_cols = ['L1_attitude', 'L2_friend', 'L3_family', 'L4_work', 'L5_visit', 'L6_lawsuit', 'collection_freq_level']

# 按论文命名定义特征集
X_A0 = df[p_cols + e_cols].copy()
X_A1 = df[p_cols + e_cols + ['sensitive_word_density']].copy()
X_A2 = df[p_cols + e_cols + ['ADJ_Ratio', 'SV_Ratio']].copy()
X_A3 = df[p_cols + e_cols + ['sensitive_word_density', 'ADJ_Ratio', 'SV_Ratio']].copy()

y = df['Spillover_flag'].copy()

print(f"\n特征数量：")
print(f"  A0 (基线): {X_A0.shape[1]}")
print(f"  A1 (+敏感词): {X_A1.shape[1]}")
print(f"  A2 (+LCM抽象): {X_A2.shape[1]}")
print(f"  A3 (+敏感词+LCM): {X_A3.shape[1]}")

def train_model(X, y, model_name):
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.3, random_state=42, stratify=y
    )
    print(f"\n{model_name} | 训练集: {len(X_train)}, 测试集: {len(X_test)}")
    model = lgb.LGBMClassifier(
        n_estimators=200,
        learning_rate=0.05,
        max_depth=5,
        num_leaves=31,
        min_child_samples=20,
        subsample=0.8,
        colsample_bytree=0.8,
        class_weight='balanced',
        random_state=42,
        verbose=-1
    )
    model.fit(
        X_train, y_train,
        eval_set=[(X_test, y_test)],
        callbacks=[lgb.early_stopping(20), lgb.log_evaluation(0)]
    )
    y_pred = model.predict(X_test)
    y_prob = model.predict_proba(X_test)[:, 1]
    return {
        'model_name': model_name,
        'f1': round(f1_score(y_test, y_pred), 4),
        'precision': round(precision_score(y_test, y_pred), 4),
        'recall': round(recall_score(y_test, y_pred), 4),
        'auc': round(roc_auc_score(y_test, y_prob), 4)
    }

# 运行 RQ A 实验
results = [
    train_model(X_A0, y, "A0_基线"),
    train_model(X_A1, y, "A1_+敏感词"),
    train_model(X_A2, y, "A2_+LCM抽象"),
    train_model(X_A3, y, "A3_+敏感词+LCM")
]

# 汇总并输出
summary = pd.DataFrame(results)
print("\n" + "="*60)
print("RQ A 增量预测效力结果（对应论文表4）")
