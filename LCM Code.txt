# ===================== 1. 催收力度（六类行为标志） =====================
L1_patterns = [
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L2_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),

]
L3_patterns = [

    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L4_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]
L5_patterns = [
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD$)', re.I),
]
L6_patterns = [
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
    re.compile(r'(SENSITIVE_WORD).*(SENSITIVE_WORD).*(SENSITIVE_WORD)', re.I),
]

def get_collection_flags(text):
    """返回六类催收行为的0/1标志"""
    if not isinstance(text, str):
        text = str(text)
    return {
        'L1_attitude': int(any(p.search(text) for p in L1_patterns)),
        'L2_friend': int(any(p.search(text) for p in L2_patterns)),
        'L3_family': int(any(p.search(text) for p in L3_patterns)),
        'L4_work': int(any(p.search(text) for p in L4_patterns)),
        'L5_visit': int(any(p.search(text) for p in L5_patterns)),
        'L6_lawsuit': int(any(p.search(text) for p in L6_patterns)),
    }

# ===================== 2. 催收频次等级（优化版） =====================
def get_frequency_level(text):
    if not isinstance(text, str):
        text = str(text)
    # 数字匹配（限定上下文）
    digit_patterns = [
        re.compile(r'(?:SENSITIVE_WORD)?\s*(?:SENSITIVE_WORD)\s*(\d+)\s*(?:SENSITIVE_WORD)?\s*SENSITIVE_WORD', re.I),
        re.compile(r'(?:SENSITIVE_WORD)?\s*(\d+)\s*(?:SENSITIVE_WORD)?\s*SENSITIVE_WORD', re.I),
        re.compile(r'SENSITIVE_WORD?(\d+)\s*SENSITIVE_WORD', re.I),
        re.compile(r'SENSITIVE_WORD(\d+)\s*SENSITIVE_WORD', re.I),
        re.compile(r'(\d+)\s*SENSITIVE_WORD', re.I),
    ]
    max_num = 0
    for pat in digit_patterns:
        m = pat.search(text)
        if m:
            # 排除自我否定语境（简略版）
            start, end = m.span()
            preceding = text[max(0, start-15):start]
            if re.search(r'(SENSITIVE_WORD)', preceding, re.I):
                continue
            num = int(m.group(1))
            if num > max_num:
                max_num = num
    if max_num > 0:
        if max_num <= 5:
            return 1
        elif max_num <= 15:
            return 2
        else:
            return 3
    # 模糊关键词
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 3
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 2
    if re.search(r'(SENSITIVE_WORD)', text, re.I):
        return 1
    # 兜底：“SENSITIVE_WORD”无数字 → 2；“SENSITIVE_WORD”无数字 → 1
    if re.search(r'(SENSITIVE_WORD).*打', text, re.I) or re.search(r'(SENSITIVE_WORD).*SENSITIVE_WORD', text, re.I):
        return 2
    if re.search(r'SENSITIVE_WORD.*SENSITIVE_WORD', text, re.I) or re.search(r'SENSITIVE_WORD.*SENSITIVE_WORD', text, re.I):
        return 1
    return 0

# ===================== 3. 敏感词（B1A，） =====================
core_sensitive_words = [
    SENSITIVE_WORD
]
escaped_words = [re.escape(w) for w in core_sensitive_words]
sensitive_pattern = re.compile('|'.join(escaped_words), re.I)

def get_sensitive_density(text):
    """返回敏感词密度（次数/总字符数 * 100）"""
    if not isinstance(text, str):
        text = str(text)
    if len(text) == 0:
        return 0.0
    count = len(sensitive_pattern.findall(text))
    return (count / len(text)) * 100

# ===================== Prompt ======================
LCM_PROMPT = """
你是一个文本分析工具。请严格按照下面的要求分析用户输入的文本，**只输出一行 JSON**，不要输出任何其他解释或标记。

### 任务
统计文本中以下两类语言特征的数量：

1. **负向形容词（ADJ）**：描述催收方/平台的负面**稳定特质**，例如：无耻、恶心、暴力、恶劣、垃圾（作形容词时）、可恶。
   - 需要判断指向对象是否为「催收方、平台、公司、你们、美团」等（计入 ADJ_to_creditor）。
   - **注意**：描述“情绪状态”的词（如崩溃、愤怒、绝望、烦躁）无论是否被副词修饰，一律归入 SV（见下方），不归入 ADJ。

2. **负向状态动词（SV）**：表达因对方行为或处境而引发的**负面情绪状态**。例如：恨、讨厌、反感、愤怒、崩溃、绝望、受不了、烦、生气、痛苦。
   - **不需要判断指向对象**，只要词语本身表达了负面情绪状态，即计入 SV_negative_total。
   - 包含“让/使/令…崩溃/愤怒”这类短语中的核心词。
   - **归属优先级**：当一个词既可作形容词又可作情绪动词时（如“崩溃”“愤怒”），一律归入 SV，不归入 ADJ。

### 不需要统计的类别
- 描述性行动动词（如“打电话”“还钱”）
- 解释性行动动词（如“骚扰”“威胁”“欺骗”）
- 其他中性或正向词汇

### 输出格式（严格 JSON）
{{
  "ADJ_total": 数字,
  "ADJ_to_creditor": 数字,
  "SV_negative_total": 数字
}}

### Few-shot 示例

def extract_json_from_response(response_text: str) -> dict:
    """从LLM返回的文本中提取JSON对象"""
    try:
        return json.loads(response_text.strip())
    except:
        pass
    match = re.search(r'\{[^{}]*\}', response_text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group())
        except:
            pass
    # 默认返回移除了 ADJ_intense_to_creditor
    return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}


def get_lcm_counts(text: str) -> dict:
    """调用LLM获取LCM统计计数"""
    if not text or len(text.strip()) < 10:
        return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}
    if len(text) > MAX_TEXT_LENGTH:
        text = text[:MAX_TEXT_LENGTH]

    prompt = LCM_PROMPT.format(text=text)

    for attempt in range(RETRY_TIMES):
        try:
            response = client.chat.completions.create(
                model=MODEL_NAME,
                messages=[{"role": "user", "content": prompt}],
                temperature=0,#0.1
                max_tokens=300,
            )
            content = response.choices[0].message.content
            counts = extract_json_from_response(content)
          
            for key in ["ADJ_total", "ADJ_to_creditor", "SV_negative_total"]:
                if key not in counts:
                    counts[key] = 0
            return counts
        except Exception as e:
            print(f"  请求失败 (尝试 {attempt + 1}/{RETRY_TIMES}): {str(e)[:50]}")
            time.sleep(2)
    return {"ADJ_total": 0, "ADJ_to_creditor": 0, "SV_negative_total": 0}


def compute_b1b(counts: dict, text: str) -> float:
    """计算 B1B 分数，"""
    total_chars = len(text)
    if total_chars == 0:
        return 0.0
    # 只保留 ADJ_to_creditor * 4 和 SV * 3
    weighted_sum = counts["ADJ_to_creditor"] * 4 + counts["SV_negative_total"] * 3
    return (weighted_sum / total_chars) * 100


def main():
    # 请根据实际数据位置修改
    input_file = "./data/打分数据集_cleaned.xlsx"

        print(f"错误：Excel文件中未找到列 '{text_column}'。可用的列有：{list(df.columns)}")
        return

    results = []
    for idx, row in df.iterrows():
        text = str(row[text_column]) if pd.notna(row[text_column]) else ""
        if len(text.strip()) < 10:
            print(f"跳过第{idx + 1}行：文本过短")
            results.append({
                "客户ID": row.get("客户ID", idx + 1),
                "ADJ_total": 0,
                "ADJ_to_creditor": 0,
                "SV_negative_total": 0,
                "B1B_score": 0,
                "text_length": len(text)
            })
            continue

        print(f"处理第{idx + 1}行，文本长度: {len(text)}")
        counts = get_lcm_counts(text)
        b1b = compute_b1b(counts, text)
        results.append({
            "客户ID": row.get("客户ID", idx + 1),
            "ADJ_total": counts["ADJ_total"],
            "ADJ_to_creditor": counts["ADJ_to_creditor"],
            "SV_negative_total": counts["SV_negative_total"],
            "B1B_score": b1b,
            "text_length": len(text)
        })
        time.sleep(SLEEP_TIME)

  
    output_file = "./output/LCM统计结果.xlsx"  
    pd.DataFrame(results).to_excel(output_file, index=False)
    print(f"\n✅ 处理完成！共处理 {len(results)} 条文本，结果保存至：{output_file}")


if __name__ == "__main__":
    main()

# ===================== 核心工具函数 =====================
def split_overlong_text(text: str, max_len: int = MAX_TEXT_LENGTH) -> str:
    """
    自动切分超长文本 
    """
    text = str(text).strip()
    if len(text) <= max_len:
        return text
    # 超长则截断，保留前max_len字符
    return text[:max_len]

def get_weights_by_length(text: str) -> dict:
    word_count = len(text)
    return weights_300_1000 if word_count <= 1000 else weights_full

def format_weights_text(weights: dict) -> str:
    sorted_items = sorted(weights.items(), key=lambda x: x[1], reverse=True)
    return "\n".join([f"- {word}（权重 {round(weight, 1)}）" for word, weight in sorted_items])

def clean_response(res: str) -> str:
    """清洗模型响应，去除标点、空格、特殊字符"""
    return re.sub(r'[^\u4e00-\u9fa5]', '', res.strip())

def map_score(response: str) -> int:
    """将文本响应映射为分数"""
    option_map = {
        "非常不同意": 1, "不同意": 2, "中立": 3, "同意": 4, "非常同意": 5
    }
    return option_map.get(response, 3)  # 默认中立

#Code for the dose-response analysis in RQ A (three-group gradient comparison)

import pandas as pd
import numpy as np
from scipy.stats import chi2_contingency
from statsmodels.stats.proportion import proportion_confint
import statsmodels.api as sm
import math
import os
import warnings

warnings.filterwarnings('ignore')

# ==================== 1. 读取数据 ====================
file_path = r""
df = pd.read_excel(file_path)

print("=" * 70)
print("【效应量计算：基于信号完整性的三组对比】")
print("=" * 70)

# ==================== 2. 构造三组 ====================
df['has_abstract'] = ((df['ADJ_Ratio'] > 0) | (df['SV_Ratio'] > 0)).astype(int)
df['has_channel'] = (df['sensitive_word_density'] > 0).astype(int)

df['signal_tier'] = '无信号'
df.loc[(df['has_abstract'] == 1) & (df['has_channel'] == 0), 'signal_tier'] = '残缺信号(IF only)'
df.loc[(df['has_abstract'] == 0) & (df['has_channel'] == 1), 'signal_tier'] = '残缺信号(THEN only)'
df.loc[(df['has_abstract'] == 1) & (df['has_channel'] == 1), 'signal_tier'] = '完整信号'

df['signal_group'] = df['signal_tier'].replace({
    '残缺信号(IF only)': '残缺信号',
    '残缺信号(THEN only)': '残缺信号'
})

# ==================== 3. 辅助函数 ====================
def cohens_h(p1, p2):
    if p1 == p2:
        return 0.0
    h1 = 2 * math.asin(math.sqrt(p1))
    h2 = 2 * math.asin(math.sqrt(p2))
    return h1 - h2


def risk_ratio(p1, p2):
    if p2 == 0:
        return np.nan
    return p1 / p2


def risk_difference_ci_newcombe(p1, n1, p2, n2, alpha=0.05):
    """
    计算两个独立比例差值的 Newcombe 置信区间。
    Newcombe, R. G. (1998). Statistics in Medicine, 17(8), 873-890.
    """
    l1, u1 = proportion_confint(int(round(n1 * p1)), n1, alpha=alpha, method='wilson')
    l2, u2 = proportion_confint(int(round(n2 * p2)), n2, alpha=alpha, method='wilson')
    diff = p1 - p2
    ci_lower = diff - np.sqrt((p1 - l1) ** 2 + (u2 - p2) ** 2)
    ci_upper = diff + np.sqrt((u1 - p1) ** 2 + (p2 - l2) ** 2)
    return diff, ci_lower, ci_upper


# ==================== 4. 分组统计 ====================
tier_order = ['无信号', '残缺信号', '完整信号']
summary = []
for tier in tier_order:
    group = df[df['signal_group'] == tier]
    n = len(group)
    complaints = group['Spillover_flag'].sum()
    p = group['Spillover_flag'].mean()
    ci_low, ci_high = proportion_confint(complaints, n, alpha=0.05, method='wilson')
    summary.append({
        'signal_group': tier,
        'n': n,
        'complaints': int(complaints),
        'complaint_rate': round(p, 4),
        'CI_low': round(ci_low, 4),
        'CI_high': round(ci_high, 4)
    })

print("\n【三组原始统计】")
print("-" * 70)
print(f"{'信号组':<12} {'样本量':>10} {'投诉数':>10} {'投诉率':>12} {'95% CI':>20}")
print("-" * 70)
for s in summary:
    ci_str = f"({s['CI_low']:.3f}, {s['CI_high']:.3f})"
    print(f"{s['signal_group']:<12} {s['n']:>10} {s['complaints']:>10} {s['complaint_rate']:>12.3f} {ci_str:>20}")

# ==================== 5. 两两对比 ====================
print("\n" + "=" * 70)
print("【两两对比】")
print("=" * 70)

comparisons = [
    ('无信号', '残缺信号'),
    ('残缺信号', '完整信号'),
    ('无信号', '完整信号')
]

results = []
for tier1, tier2 in comparisons:
    g1 = df[df['signal_group'] == tier1]
    g2 = df[df['signal_group'] == tier2]

    n1, n2 = len(g1), len(g2)
    p1, p2 = g1['Spillover_flag'].mean(), g2['Spillover_flag'].mean()

    rr = risk_ratio(p2, p1) if p1 > 0 else np.nan
    diff, diff_low, diff_high = risk_difference_ci_newcombe(p1, n1, p2, n2)
    h = cohens_h(p1, p2)

    results.append({
        'comparison': f"{tier1} → {tier2}",
        'rate_1': round(p1, 4),
        'n_1': n1,
        'rate_2': round(p2, 4),
        'n_2': n2,
        'risk_difference': round(diff, 4),
        'RD_CI_low': round(diff_low, 4),
        'RD_CI_high': round(diff_high, 4),
        'risk_ratio': round(rr, 4) if not np.isnan(rr) else np.nan,
        'cohens_h': round(h, 4)
    })

print(
    f"\n{'对比':<20} {'投诉率1':>12} {'投诉率2':>12} {'风险差':>12} {'95% CI':>22} {'风险比(RR)':>12} {'Cohen_h':>12}")
print("-" * 110)
for r in results:
    ci_str = f"({r['RD_CI_low']:.3f}, {r['RD_CI_high']:.3f})"
    print(
        f"{r['comparison']:<20} {r['rate_1']:>12.3f} {r['rate_2']:>12.3f} {r['risk_difference']:>12.3f} {ci_str:>22} {r['risk_ratio']:>12.3f} {r['cohens_h']:>12.3f}")

# ==================== 6. 趋势检验 ====================
print("\n" + "=" * 70)
print("【趋势检验】")
print("=" * 70)

order_map = {'无信号': 0, '残缺信号': 1, '完整信号': 2}
df['signal_ordinal'] = df['signal_group'].map(order_map)

X_trend = df[['signal_ordinal']].values
X_trend = sm.add_constant(X_trend)
model_trend = sm.Logit(df['Spillover_flag'], X_trend).fit(disp=False)

trend_coef = model_trend.params[1]
trend_p = model_trend.pvalues[1]

print(f"趋势检验（有序0→1→2，Logistic回归）：")
print(f"  系数: {trend_coef:.4f}")
print(f"  p值: {trend_p:.6f}")
print("  投诉率随信号完整性呈单调递增趋势（显著）" if trend_p < 0.05 else "  趋势不显著")

# ==================== 7. 汇总（论文可直接引用） ====================
print("\n" + "=" * 70)
print("【论文可直接引用的结果】")
print("=" * 70)

no_signal = df[df['signal_group'] == '无信号']
partial = df[df['signal_group'] == '残缺信号']
full = df[df['signal_group'] == '完整信号']

p_none = no_signal['Spillover_flag'].mean()
p_partial = partial['Spillover_flag'].mean()
p_full = full['Spillover_flag'].mean()
n_none, n_partial, n_full = len(no_signal), len(partial), len(full)

ci_low_none, ci_high_none = proportion_confint(no_signal['Spillover_flag'].sum(), n_none, method='wilson')
ci_low_partial, ci_high_partial = proportion_confint(partial['Spillover_flag'].sum(), n_partial, method='wilson')
ci_low_full, ci_high_full = proportion_confint(full['Spillover_flag'].sum(), n_full, method='wilson')

print(f"\n无信号组: {p_none:.3f} (95% CI: {ci_low_none:.3f}-{ci_high_none:.3f})")
print(f"残缺信号组: {p_partial:.3f} (95% CI: {ci_low_partial:.3f}-{ci_high_partial:.3f})")
print(f"完整信号组: {p_full:.3f} (95% CI: {ci_low_full:.3f}-{ci_high_full:.3f})")

rr = p_full / p_none if p_none > 0 else np.nan
rr_ci_low, rr_ci_high = np.nan, np.nan
if not np.isnan(rr) and rr > 0:
    log_rr = math.log(rr)
    se_log_rr = math.sqrt(
        1 / full['Spillover_flag'].sum() - 1 / n_full + 1 / no_signal['Spillover_flag'].sum() - 1 / n_none)
    rr_ci_low = math.exp(log_rr - 1.96 * se_log_rr)
    rr_ci_high = math.exp(log_rr + 1.96 * se_log_rr)
    print(f"\n风险比（完整 vs 无信号）: {rr:.2f} 倍 (95% CI: {rr_ci_low:.2f} - {rr_ci_high:.2f})")
else:
    print("\n风险比无法计算（某组投诉率为0或负值）")

print(f"\n所有结果已导出至: {output_path}")

file_path = r""

df['treatment'] = ((df['sensitive_word_density'] > 0) &

print(f"处理组匹配前投诉率: {treatment_before_rate:.3f} ({treatment_before_rate * 100:.1f}%)")
print(f"对照组匹配前投诉率: {control_before_rate:.3f} ({control_before_rate * 100:.1f}%)")

# ============================================================
# 2. 倾向得分估计
# ============================================================
X = df[covariates].copy()
y = df['treatment'].copy()
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

ps_model = LogisticRegression(max_iter=1000, random_state=42)
ps_model.fit(X_scaled, y)
df['ps'] = ps_model.predict_proba(X_scaled)[:, 1]
df['ps_logit'] = np.log(df['ps'] / (1 - df['ps'] + 1e-8))
print(f"ps_logit标准差: {df['ps_logit'].std():.4f}")
print(f"卡尺值 (Logit尺度的0.2倍): {0.2 * df['ps_logit'].std():.4f}")

# ============================================================
# 3. PSM匹配（使用 ps_logit）
# ============================================================
treatment_df = df[df['treatment'] == 1].copy().sort_values('ps_logit').reset_index(drop=True)
control_df = df[df['treatment'] == 0].copy().sort_values('ps_logit').reset_index(drop=True)

caliper = 0.2 * df['ps_logit'].std()
print(f"\n卡尺阈值 (logit尺度): {caliper:.4f}")

k_neighbors = 5
tree = KDTree(control_df['ps_logit'].values.reshape(-1, 1))
distances, indices = tree.query(treatment_df['ps_logit'].values.reshape(-1, 1), k=k_neighbors)

matched_pairs = []
used_control_idx = set()
discarded_indices = []

for i in range(len(treatment_df)):
    sorted_pairs = sorted(zip(distances[i], indices[i]), key=lambda x: x[0])
    matched = False
    for dist, c_idx in sorted_pairs:
        if dist > caliper:
            break
        if c_idx not in used_control_idx:
            matched_pairs.append((i, c_idx))
            used_control_idx.add(c_idx)
            matched = True
            break
    if not matched:
        discarded_indices.append(i)

matched_treatment = treatment_df.iloc[[p[0] for p in matched_pairs]].reset_index(drop=True)
matched_control = control_df.iloc[[p[1] for p in matched_pairs]].reset_index(drop=True)
discarded_treatment = treatment_df.iloc[discarded_indices].reset_index(drop=True)
matched_df = pd.concat([matched_treatment, matched_control], ignore_index=True)

print(f"\n成功匹配: {len(matched_pairs)} 对")
print(f"丢弃处理组: {len(discarded_treatment)} 例")

# 4. SMD平衡性检验
# ============================================================
def compute_smd(df, treatment_col, cols):
    treated = df[df[treatment_col] == 1]
    control = df[df[treatment_col] == 0]
    smd_dict = {}
    for col in cols:
        if col in df.columns:
            m1, m2 = treated[col].mean(), control[col].mean()
            v1, v2 = treated[col].var(), control[col].var()
            pooled = np.sqrt((v1 + v2) / 2)
            smd_dict[col] = abs((m1 - m2) / pooled) if pooled > 0 else 0
    return smd_dict

smd_before = compute_smd(df, 'treatment', covariates)
smd_after = compute_smd(matched_df, 'treatment', covariates)

print("\n" + "=" * 70)
print("【表E】SMD平衡性检验")

def bootstrap_att(treatment_out, control_out, n_bootstrap=1000, seed=42):
    np.random.seed(seed)
    n_pairs = len(treatment_out)
    atts = []
    for _ in range(n_bootstrap):
        idx = np.random.choice(n_pairs, size=n_pairs, replace=True)
        t_sample = treatment_out[idx]
        c_sample = control_out[idx]
        atts.append(t_sample.mean() - c_sample.mean())
    lower = np.percentile(atts, 2.5)
    upper = np.percentile(atts, 97.5)
    return lower, upper, atts

def random_subsample_sensitivity(matched_treat, matched_ctrl, n_repeats=200, fraction=2/3, seed=42):
    np.random.seed(seed)
    n_pairs = len(matched_treat)
    atts = []
    p_vals = []
    n_signif = 0
    for _ in range(n_repeats):
        idx = np.random.choice(n_pairs, size=int(n_pairs * fraction), replace=False)
        t_sub = matched_treat.iloc[idx]['Spillover_flag'].values
        c_sub = matched_ctrl.iloc[idx]['Spillover_flag'].values
        att = t_sub.mean() - c_sub.mean()
        atts.append(att)
        b = ((t_sub == 0) & (c_sub == 1)).sum()
        c = ((t_sub == 1) & (c_sub == 0)).sum()
        if b + c > 0:
            p = 2 * binom.cdf(min(b, c), b + c, 0.5)
            if p > 1:
                p = 2 * (1 - binom.cdf(min(b, c) - 1, b + c, 0.5))
        else:
            p = 1.0
        p_vals.append(p)
        if p < 0.05:
            n_signif += 1

    atts = np.array(atts)
    p_vals = np.array(p_vals)
    result = {
        'n_repeats': n_repeats,
        'fraction': fraction,
        'ATT_mean': atts.mean(),
        'ATT_median': np.median(atts),
        'ATT_2.5%': np.percentile(atts, 2.5),
        'ATT_97.5%': np.percentile(atts, 97.5),
        'p_mean': p_vals.mean(),
        'p_median': np.median(p_vals),
        'p_2.5%': np.percentile(p_vals, 2.5),
        'p_97.5%': np.percentile(p_vals, 97.5),
        'signif_proportion': n_signif / n_repeats
    }
    return result, atts, p_vals

subsample_result, atts_subsample, p_vals_subsample = random_subsample_sensitivity(

def design_effect_sensitivity(p_original, label, k_values=[1.0, 1.45, 1.5, 2.0, 2.5, 3.0]):
    """
    设计效应敏感性分析。
    p_original: 原始McNemar p值
    label: 模型名称
    k_values: 每个客户平均出现次数的假设值
    """
    z_original = norm.ppf(1 - p_original / 2)

    print(f"\n{'='*60}")
    print(f"设计效应敏感性分析：{label}")
    print(f"原始p值 = {p_original:.4f}，原始z值 = {z_original:.4f}")
    print(f"{'='*60}")
    print(f"{'k':<8} {'DEFF':<8} {'SE放大':<10} {'调整后z':<10} {'调整后p':<10} {'显著?':<8}")
    print("-" * 60)

    results = []
    for k in k_values:
        deff = k
        se_multiplier = np.sqrt(deff)
        z_adjusted = z_original / se_multiplier
        p_adjusted = 2 * (1 - norm.cdf(z_adjusted))
        sig = "✅" if p_adjusted < 0.05 else "❌"
        print(f"{k:<8.1f} {deff:<8.2f} {se_multiplier:<10.2f} {z_adjusted:<10.4f} {p_adjusted:<10.4f} {sig:<8}")
        results.append({
            '模型': label,
            'k': k,
            'DEFF': deff,
            'SE放大倍数': se_multiplier,
            '调整后z': z_adjusted,
            '调整后p': p_adjusted,
            '是否显著': p_adjusted < 0.05
        })
    return results

gamma_values = [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]

def rosenbaum_sensitivity(t_out, c_out, gamma_values):
    diffs = t_out - c_out
    n = len(diffs)
    abs_diffs = np.abs(diffs)
    ranks = np.argsort(abs_diffs) + 1
    signs = np.sign(diffs)
    results = []
    for gamma in gamma_values:
        w_plus = np.sum(ranks * (signs == 1))
        w_minus = np.sum(ranks * (signs == -1))
        w_stat = min(w_plus, w_minus)
        if n > 0 and np.var(diffs) > 0:
            try:
                _, p_val = wilcoxon(diffs, correction=True)
            except:
                p_val = 1.0
        else:
            p_val = 1.0
        p_upper = min(1.0, p_val * gamma)
        results.append({'gamma': gamma, 'p_upper': p_upper})
    return results

# -*- coding: utf-8 -*-

import os

import pandas as pd
from tqdm import tqdm
from openai import OpenAI

# ==================== 配置区 ====================
INPUT_PATH = r""
OUTPUT_PATH = r""

# DeepSeek API 配置

API_KEY = ""   # 填你的 Access Token
BASE_URL = "https://api.deepseek.com"
MODEL = "deepseek-v4-flash"   # 如果实际模型名不同，请修改
TEMPERATURE = 0
MAX_RETRIES = 3

client = OpenAI(api_key=API_KEY, base_url=BASE_URL)

# ==================== 工具函数 ====================
def extract_json(text: str):
    """从模型返回内容中提取 JSON，兼容 markdown 代码块和夹杂文字。"""
    text = text.strip()
    text = re.sub(r"^```(?:json)?\s*", "", text, flags=re.M)
    text = re.sub(r"\s*```$", "", text, flags=re.M)
    try:
        return json.loads(text)
    except Exception:
        pass
    match = re.search(r"\{.*\}", text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group())
        except Exception:
            pass
    raise ValueError(f"无法解析 JSON，模型原始输出：{text[:500]}")

def score_one_text(text: str):
    """
    调用 DeepSeek 对单条中文文本进行满意度打分。
    返回：(score, confidence, explanation)
    """
    if not isinstance(text, str) or not text.strip():
        return None, None, "空文本，跳过"

    # 按论文 Respondent 思路改写的中文 prompt
    prompt = f"""你是一个专业的用户体验与满意度分析专家。请根据以下客户反馈文本，对客户的满意度进行整体打分。

要求：
1. 输出一个整体满意度分数，范围 0.00 到 1.00，保留两位小数。0.00 表示非常不满意，1.00 表示非常满意。
2. 给出你对该分数的置信度，只能是 low、medium、high 之一。
3. 给出一段详细的中文解释，说明为什么给出这个分数，要能让非技术人员理解。
4. 只输出一个 JSON 对象，不要有其他内容。格式如下：
{{"score": 0.85, "confidence": "high", "explanation": "客户对产品整体表示满意，特别提到了……"}}

客户反馈文本：
{text}
"""

    last_error = None
    for attempt in range(MAX_RETRIES):
        try:
            response = client.chat.completions.create(
                model=MODEL,
                messages=[{"role": "user", "content": prompt}],
                temperature=TEMPERATURE,
                # 关键：关闭 thinking，防止打分趋向均值
                extra_body={"thinking": {"type": "disabled"}}
            )
            content = response.choices[0].message.content
            data = extract_json(content)
            score = float(data.get("score"))
            score = max(0.0, min(1.0, score))
            confidence = str(data.get("confidence", "")).lower()
            explanation = str(data.get("explanation", ""))
            return score, confidence, explanation
        except Exception as e:
            last_error = e
            time.sleep(1 + attempt)

    return None, None, f"失败：{last_error}"

# ==================== 主流程 ====================
def main():
    print("读取 Excel ...")
    df = pd.read_excel(INPUT_PATH)

    # 注意：如果你的 text 列名不是 "text"，请改成实际列名
    TEXT_COL = "text"
    if TEXT_COL not in df.columns:
        raise ValueError(f"Excel 中未找到 '{TEXT_COL}' 列，请检查表头。当前列名：{list(df.columns)}")

    print(f"共读取 {len(df)} 行数据，开始打分 ...")

    scores = []
    confidences = []
    explanations = []

    for idx, row in tqdm(df.iterrows(), total=len(df), desc="满意度打分"):
        text = row[TEXT_COL]
        score, confidence, explanation = score_one_text(text)
        scores.append(score)
        confidences.append(confidence)
        explanations.append(explanation)

    df["sentiment_score"] = scores
    df["confidence"] = confidences
    df["explanation"] = explanations

    print(f"保存结果到：{OUTPUT_PATH}")
    df.to_excel(OUTPUT_PATH, index=False)
    print("完成。")

if __name__ == "__main__":
    main()
